import java.util.ArrayList;
import java.util.List;

// =======================================================
// SearchScreen
// =======================================================

class SearchScreen {

    // Association
    private ManageProductClass manageProductClass;

    public SearchScreen(
            ManageProductClass manageProductClass) {

        this.manageProductClass =
                manageProductClass;
    }

    public void display() {

        System.out.println(
                "===== SEARCH SCREEN =====");

        System.out.println(
                "1. Product Management");
    }

    public void selectProductManage() {

        System.out.println(
                "Producer selected Product Management");

        manageProductClass.searchProducts();
    }
}

// =======================================================
// ManageProductClass
// =======================================================

class ManageProductClass {

    // Associations
    private SearchScreen searchScreen;

    private AvailableProductsScreen
            availableProductsScreen;

    private DataEntryModificationScreen
            dataEntryModificationScreen;

    private DBManager dbManager;

    public void init() {

        System.out.println(
                "Initializing Product Management...");

        this.dbManager = new DBManager();

        this.searchScreen =
                new SearchScreen(this);

        searchScreen.display();
    }

    public void searchProducts() {

        System.out.println(
                "\nSearching Products...");

        List<Product> products =
                dbManager.readProducts();

        availableProductsScreen =
                new AvailableProductsScreen(this);

        availableProductsScreen
                .showProducts(products);
    }

    public void openProductForm() {

        dataEntryModificationScreen =
                new DataEntryModificationScreen(this);

        dataEntryModificationScreen
                .enterProductDetails();
    }

    public void saveProduct(Product product) {

        System.out.println(
                "\nSaving Product...");

        boolean validQuantity =
                dbManager
                .validateProductDetails(product);

        if (!validQuantity) {

            InvalidQuantityScreen screen =
                    new InvalidQuantityScreen();

            screen.display();

            return;
        }

        boolean validPricing =
                dbManager
                .checkPricingRules(product);

        if (!validPricing) {

            InvalidPricingScreen screen =
                    new InvalidPricingScreen();

            screen.display();

            return;
        }

        dbManager.saveProduct(product);

        ValidEntryNotificationScreen screen =
                new ValidEntryNotificationScreen();

        screen.display();
    }
}

// =======================================================
// AvailableProductsScreen
// =======================================================

class AvailableProductsScreen {

    // Association
    private ManageProductClass manageProductClass;

    public AvailableProductsScreen(
            ManageProductClass manageProductClass) {

        this.manageProductClass =
                manageProductClass;
    }

    public void showProducts(
            List<Product> products) {

        System.out.println(
                "\n===== AVAILABLE PRODUCTS =====");

        for (Product product : products) {

            System.out.println(
                    "Product: "
                            + product.getProductName()
                            + " | Quantity: "
                            + product.getQuantity());
        }

        addProduct();
    }

    private void addProduct() {

        System.out.println(
                "\nProducer selected Add Product");

        manageProductClass.openProductForm();
    }

    private void editProduct() {

        System.out.println(
                "\nProducer selected Edit Product");
    }
}

// =======================================================
// DataEntryModificationScreen
// =======================================================

class DataEntryModificationScreen {

    // Associations
    private ManageProductClass manageProductClass;

    private Product product;

    public DataEntryModificationScreen(
            ManageProductClass manageProductClass) {

        this.manageProductClass =
                manageProductClass;
    }

    public void enterProductDetails() {

        System.out.println(
                "\n===== PRODUCT ENTRY SCREEN =====");

        System.out.println(
                "Entering Product Details...");

        String productName = "Tomatoes";

        int quantity = 50;

        String pricingRules =
                "1-10 : 5€, 11-20 : 4€";

        product =
                new Product(
                        productName,
                        quantity,
                        pricingRules);

        setPricingRules();

        submitProduct();
    }

    public void modifyProductDetails() {

        System.out.println(
                "Modifying Product Details...");
    }

    public void setPricingRules() {

        System.out.println(
                "Setting Pricing Rules...");
    }

    public void submitProduct() {

        manageProductClass.saveProduct(product);
    }
}

// =======================================================
// Product
// =======================================================

class Product {

    // Attributes
    private String productName;

    private int quantity;

    private String pricingRules;

    public Product(
            String productName,
            int quantity,
            String pricingRules) {

        this.productName = productName;

        this.quantity = quantity;

        this.pricingRules = pricingRules;
    }

    public String getProductName() {
        return productName;
    }

    public int getQuantity() {
        return quantity;
    }

    public String getPricingRules() {
        return pricingRules;
    }

    public void setProductDetails(
            String productName,
            int quantity,
            String pricingRules) {

        this.productName = productName;

        this.quantity = quantity;

        this.pricingRules = pricingRules;
    }

    public void getProductDetails() {

        System.out.println(
                "Product: " + productName);

        System.out.println(
                "Quantity: " + quantity);

        System.out.println(
                "Pricing Rules: "
                        + pricingRules);
    }
}

// =======================================================
// DBManager
// =======================================================

class DBManager {

    public List<Product> readProducts() {

        System.out.println(
                "Reading products from database...");

        List<Product> products =
                new ArrayList<>();

        products.add(
                new Product(
                        "Potatoes",
                        100,
                        "1-10 : 3€"));

        products.add(
                new Product(
                        "Tomatoes",
                        50,
                        "1-10 : 5€"));

        return products;
    }

    public boolean validateProductDetails(
            Product product) {

        System.out.println(
                "Validating Quantity...");

        return product.getQuantity() >= 0;
    }

    public boolean checkPricingRules(
            Product product) {

        System.out.println(
                "Checking Pricing Rules...");

        return product
                .getPricingRules() != null;
    }

    public void saveProduct(Product product) {

        System.out.println(
                "Saving Product to Database...");
    }
}
