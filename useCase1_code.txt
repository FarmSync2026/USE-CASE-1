import java.util.ArrayList;

// 1. ENTITY LAYER (Επίπεδο Οντοτήτων)

class Product {
    private String id;
    private String name;
    private double quantity;
    private String pricingRules;

    public Product(String id, String name, double quantity, String pricingRules) {
        this.id = id;
        this.name = name;
        this.quantity = quantity;
        this.pricingRules = pricingRules;
    }

    // Getters και Setters
    public String getId() { return id; }
    public String getName() { return name; }
    
    // Η μέθοδος getQuantity() που χρησιμοποιείται για την επικύρωση των δεδομένων
    public double getQuantity() { return quantity; }
    public String getPricingRules() { return pricingRules; }

    public void setQuantity(double quantity) { this.quantity = quantity; }
    public void setPricingRules(String pricingRules) { this.pricingRules = pricingRules; }
}



// 2. INFRASTRUCTURE / UTILITY LAYER (Υποδομή / Βάση Δεδομένων)

class DBManager {
    // Βήμα 6: Επικύρωση διαθεσιμότητας / ονόματος στη Βάση Δεδομένων
    public boolean checkAvailability(String productName) {
        System.out.println("[DBManager] Έλεγχος στη Βάση Δεδομένων για την εγκυρότητα του ονόματος: " + productName);
        return true; 
    }

    public void saveProduct(Product product) {
        System.out.println("[DBManager] SQL: INSERT INTO Products VALUES ('" 
                           + product.getId() + "', '" 
                           + product.getName() + "', " 
                           + product.getQuantity() + ", '" 
                           + product.getPricingRules() + "');");
        System.out.println("[DBManager] Το προϊόν αποθηκεύτηκε επιτυχώς στη Βάση Δεδομένων.");
    }
}



// 3. CONTROLLER LAYER (Επίπεδο Ελεγκτών Λογικής)

clas
s ManageProductClass {
    private DBManager dbManager = new DBManager();

    // Βήμα 2: Αναζήτηση και φόρτωση υπαρχόντων προϊόντων
    public void loadProducts() {
        System.out.println("[ManageProductClass] Αναζήτηση υπαρχόντων προϊόντων στη Βάση Δεδομένων...");
    }
}


class DataEntryModificationClass {
    private DBManager dbManager = new DBManager();

    public void validateProductDetails(String name, double quantity, String pricingRules) {
        System.out.println("[DataEntryModificationClass] Έναρξη επικύρωσης στοιχείων...");

        if (quantity < 0) {
            System.out.println("[DataEntryModificationClass] Ανίχνευση λάθους: Ποσότητα < 0 (" + quantity + ")");
            NoValidQuantityScreen errorScreen = new NoValidQuantityScreen();
            errorScreen.showQuantityError();
            return; 
        }

        if (pricingRules == null || pricingRules.trim().isEmpty() || pricingRules.equalsIgnoreCase("invalid")) {
            System.out.println("[DataEntryModificationClass] Ανίχνευση λάθους: Μη έγκυροι κανόνες τιμολόγησης.");
            InvalidPricingRulesScreen errorRulesScreen = new InvalidPricingRulesScreen();
            errorRulesScreen.showPricingError();
            return; 
        }

        dbManager.checkAvailability(name);
        
        String generatedId = "PROD_" + System.currentTimeMillis();
        Product newProduct = new Product(generatedId, name, quantity, pricingRules);
        
        dbManager.saveProduct(newProduct);

        ValidEntryInformationScreen successScreen = new ValidEntryInformationScreen();
        successScreen.showSuccess();
    }
}



// 4. BOUNDARY LAYER / SCREENS (Επίπεδο Διεπαφών Χρήστη)

class SearchScreen {
    private ManageProductClass controller = new ManageProductClass();

    // Βήμα 1: Ο παραγωγός επιλέγει "Διαχείριση προϊόντων"
    public void selectProductMgmt() {
        System.out.println("[SearchScreen] Ο παραγωγός επιλέγει την 'Διαχείριση προϊόντων'.");
        controller.loadProducts();
    }
}

class DataEntryModificationScreen {
    private DataEntryModificationClass validationController = new DataEntryModificationClass();

    public void enterDetails(String name, double quantity, String pricingRules) {
        System.out.println("[DataEntryModificationScreen] Εισαγωγή -> Όνομα: " + name 
                           + " | Ποσότητα: " + quantity 
                           + " | Κανόνες: " + pricingRules);
        
        validationController.validateProductDetails(name, quantity, pricingRules);
    }
}
